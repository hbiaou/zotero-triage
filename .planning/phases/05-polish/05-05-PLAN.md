```typescript
async loadItems(onProgress?: LoadProgressCallback): Promise<ZoteroItem[]> {
  return await retryWithBackoff(async () => {
    // Query total count first
    const totalResult = this.db.exec("SELECT COUNT(*) as total FROM items WHERE itemID NOT IN (SELECT itemID FROM deletedItems)");
    const total = totalResult[0].values[0][0] as number;

    // Call progress callback initially
    if (onProgress) {
      onProgress(0, total);
    }

    // Existing chunked processing
    const allItems: ZoteroItem[] = [];
    const chunkSize = 50; // From 01-02 decision

    for (let offset = 0; offset < total; offset += chunkSize) {
      const chunk = await this.loadItemsChunk(offset, chunkSize);
      allItems.push(...chunk);

      // Report progress after each chunk
      if (onProgress) {
        onProgress(allItems.length, total);
      }
    }

    return allItems;
  }, { maxAttempts: 5, initialDelayMs: 100 });
}
```

```typescript
async generateAndShowBatch(): Promise<void> {
  const progress = new ProgressTracker();

  try {
    // Start progress tracking
    progress.start('Loading Zotero library...', 5000); // Estimate 5000 items

    // Load items with progress callback
    const items = await this.plugin.connector.loadItems((loaded, total) => {
      progress.update(loaded, `Loading items from database...`);
      // Update estimate if total differs
      if (total !== progress['state'].total) {
        progress['state'].total = total;
      }
    });

    progress.update(items.length, 'Generating batch...');

    // Generate batch
    const batch = await this.plugin.batchService.generateBatch(
      this.plugin.settings.batchSize
    );

    progress.complete(`Generated batch of ${batch.length} items`);

    // Render batch (existing logic)
    this.currentBatch = batch;
    this.currentIndex = 0;
    this.render();

  } catch (err) {
    const context = getErrorContext(err as Error);
    progress.error(context.message);
    // Show error modal for detailed actions
    new ErrorModal(this.app, context).open();
  }
}
```

```typescript
  const progress = new ProgressTracker();

  try {
    progress.start('Filtering candidates...', 100); // Arbitrary estimate

    // Get all unprocessed items
    const allItems = await this.connector.loadItems();
    const unprocessed = this.filterUnprocessed(allItems);

    progress.update(50, 'Scoring candidates...');

    // Score items (existing logic)
    const scored = this.scoreItems(unprocessed);

    progress.update(75, 'Selecting batch...');

    // Select top N items
    const batch = scored
      .sort((a, b) => b.score - a.score)
      .slice(0, batchSize)
      .map(s => s.item);

    progress.complete();

    return batch;

  } catch (err) {
    progress.error('Failed to generate batch');
    throw err; // Re-throw for upstream handling
  }
}
```
