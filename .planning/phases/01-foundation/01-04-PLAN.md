Update onload():
After existing initialization, add:

- Create NoteGenerator instance with app and settings.outputFolder

- Add command "zotbridge:import-item":
  - id: "zotbridge-import-item"
  - name: "Import Zotero item"
  - callback: async () => {
      // Check database configured
      if (!this.settings.zoteroDbPath) {
        new Notice("Please configure Zotero database path in settings");
        return;
      }

      // Load items if not loaded (with progress notice)
      let items = await this.loadItemsWithProgress();
      if (!items || items.length === 0) {
        new Notice("No items found in Zotero database");
        return;
      }

      // Open search modal
      new ItemSearchModal(this.app, items, (selectedItem) => {
        this.showPreviewAndImport(selectedItem);
      }).open();
    }
