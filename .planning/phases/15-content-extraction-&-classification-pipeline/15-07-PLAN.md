   ```typescript
   // Check evidence level
   const evidenceLevel = this.evidenceExtractor.getEvidenceLevel(item);

   if (evidenceLevel === 'MetadataOnly') {
     // Generate diagnostic note instead of attempting enrichment
     const diagnosticNote = await this.diagnosticNoteService.generateDiagnosticNote(
       item,
       this.determineDiagnosticReason(item)  // Helper function
     );

     // Store diagnostic note with appropriate metadata
     this.registry.setProcessingState(item.key, {
       state: 'enrichment_pending',
       diagnosticNote: diagnosticNote,
       reason: this.determineDiagnosticReason(item)
     });

     return;  // Don't attempt enrichment, queue for retry
   }
   ```

4. Create helper method determineDiagnosticReason():
   This method checks what evidence is missing:
   ```typescript
   private determineDiagnosticReason(item: ZoteroItem): DiagnosticReason {
     const hasPDF = !!item.pdfPath;
     const hasNotes = item.notes && item.notes.length > 0;
     const hasAbstract = !!item.abstract;
     const isVideo = ['videoRecording', 'video'].includes(item.itemType);

     if (!hasPDF && !hasNotes && !hasAbstract) {
       return 'metadata_only';
     }
     if (hasAbstract && !hasPDF && !hasNotes) {
       return 'abstract_only';
     }
     if (isVideo && !hasTranscript) {
       return 'no_transcript';
     }
     if (hasPDF && !hasNotes) {
       return 'no_notes';
     }
     if (hasNotes && !hasPDF) {
       return 'no_pdf';
     }
     return 'metadata_only';  // fallback
   }
   ```

5. Add DiagnosticReason type to extract/types.ts if not present:
   export type DiagnosticReason = 'no_pdf' | 'no_notes' | 'no_transcript' | 'abstract_only' | 'metadata_only';
